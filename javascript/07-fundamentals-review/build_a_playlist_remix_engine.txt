function flattenPlaylists(playlists) {
  if (!Array.isArray(playlists)) {
    return [];
  }
  const outputPlaylist = [...playlists];
  for (let i = 0; i < outputPlaylist.length; i++) {
    outputPlaylist[i].map((track, trackIndex) => {
      track.source = [i, trackIndex];
    });
  }
  return outputPlaylist.flat();
}

function scoreTracks(tracks) {
  tracks.map(track => {
    track.score = track.votes * 10 - Math.abs(track.bpm - 120);
  })
  return tracks;
}

function dedupeTracks(tracks) {
  const output = [];
  const existed = [];
  for (const track of tracks) {
    if (existed.includes(track.trackId)) {
      continue;
    }
    existed.push(track.trackId);
    output.push(track);
  }
  return output;
}

function enforceArtistQuota(tracks, limit) {
  const occurences = {};
  const output = [];
  tracks.forEach(track => occurences[track.artist] = 0);
  for (const track of tracks) {
    if (occurences[track.artist] === limit) {
      continue;
    }
    occurences[track.artist]++;
    output.push(track);
  }
  return output;
}

function buildSchedule(tracks) {
  return tracks.map((track, index) => { 
    return { slot: index+1, trackId: track.trackId };
  })
}

function remixPlaylist(playlists, limOfArtistOccurences) {
  return buildSchedule(enforceArtistQuota(dedupeTracks(scoreTracks(flattenPlaylists(playlists))), limOfArtistOccurences));
}
