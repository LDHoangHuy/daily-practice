/*
0 1 1 2 3 5 8 13 ...
0: -1 + 1
i: prev + i
1: 1 + 0
i: prev + i
1: 0 + 1
i: prev + i
2: 1 + 1
 */

function sumFibs(num) {
  let sum = 0;
  let a = 0;
  let b = 1;
  while (a <= num) {
    if (a % 2 !== 0) {
      sum += a;
    }
    
    const next = a + b;
    a = b;
    b = next;
  }
  return sum;
}

console.log(sumFibs(1000));
